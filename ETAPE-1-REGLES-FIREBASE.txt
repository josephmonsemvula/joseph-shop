rules_version = '2';
service cloud.firestore {
  match /databases/{database}/documents {
    function adm() { return request.auth != null && request.auth.token.email == 'joseph.admin@josselshop.app'; }
    function me(u) { return request.auth != null && request.auth.uid == u; }
    function shopOf(s) { return get(/databases/$(database)/documents/shops/$(s)).data; }

    match /shops/{s} {
      allow read: if true;
      allow create: if me(request.resource.data.uid) && request.resource.data.status == 'pending' && request.resource.data.cpaid == 0;
      allow update: if adm() || (me(resource.data.uid)
        && request.resource.data.uid == resource.data.uid
        && request.resource.data.cpaid == resource.data.cpaid
        && (request.resource.data.status == resource.data.status
            || (resource.data.status == 'pending' && request.resource.data.status == 'active'
                && request.resource.data.get('try', '') == get(/databases/$(database)/documents/codes/$(resource.data.ph9)).data.code)));
      allow delete: if adm();
    }
    match /products/{p} {
      allow read: if true;
      allow create: if me(request.resource.data.uid) && shopOf(request.resource.data.slug).uid == request.auth.uid && shopOf(request.resource.data.slug).status == 'active';
      allow update, delete: if adm() || me(resource.data.uid);
    }
    match /private/{p} {
      allow create: if me(request.resource.data.uid);
      allow read, update, delete: if adm() || me(resource.data.uid);
    }
    match /orders/{o} {
      allow get: if true;
      allow list: if adm() || (request.auth != null && resource.data.uid == request.auth.uid);
      allow create: if request.resource.data.status == 'new' && shopOf(request.resource.data.slug).status == 'active' && request.resource.data.uid == shopOf(request.resource.data.slug).uid;
      allow update: if adm() || (me(resource.data.uid) && request.resource.data.diff(resource.data).affectedKeys().hasOnly(['status','files']))
        || (resource.data.status == 'new' && request.resource.data.diff(resource.data).affectedKeys().hasOnly(['ref']));
      allow delete: if adm();
    }
    match /codes/{c} { allow read, write: if adm(); }
  }
}
